public class Exercise06_43 {

    public static void main(String[] args) {

        System.out.println("i\tm(i)");
        for (int i = 1; i <= 100; i++)
            System.out.println(i + "\t" + m(i));
    }
    public static double m(int n) {
        double sum = 0;
        for (int k = 1; k <= n; k++)
            sum += 1.0 / (k * (k + 1));
        return sum;
    }
}
