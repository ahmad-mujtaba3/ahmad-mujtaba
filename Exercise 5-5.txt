public class Exercise05_05 {

    public static void main(String[] args) {

        System.out.println("Kilograms   Pounds   |   Pounds   Kilograms");
        int p = 20;
        for (int kg = 1; kg <= 199; kg += 2, p += 5)
            System.out.printf("%-10d%7.1f   |   %-7d%10.2f%n", kg, kg * 2.2, p, p / 2.2);
    }
}
