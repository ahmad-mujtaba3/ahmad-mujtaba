import java.util.Scanner;

public class Exercise05_38 {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        String octal = "";
        while (n > 0) {
            octal = (n % 8) + octal;
            n /= 8;
        }
        System.out.println(octal);
    }
}
