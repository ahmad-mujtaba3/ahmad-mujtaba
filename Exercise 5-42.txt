import java.util.Scanner;

public class Exercise05_42 {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        double commissionWanted = input.nextDouble();
        double sales = 0, commission = 0;
        while (commission < commissionWanted) {
            sales++;
            if (sales <= 5000)
                commission = sales * 0.08;
            else if (sales <= 10000)
                commission = 5000 * 0.08 + (sales - 5000) * 0.10;
            else
                commission = 5000 * 0.08 + 5000 * 0.10 + (sales - 10000) * 0.12;
        }
        System.out.println(sales);
    }
}
