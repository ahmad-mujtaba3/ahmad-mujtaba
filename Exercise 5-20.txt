public class Exercise05_20 {

    public static void main(String[] args) {

        int count = 0;
        for (int i = 2; i < 1000; i++) {
            boolean prime = true;
            for (int j = 2; j <= i / 2; j++)
                if (i % j == 0) { prime = false; break; }
            if (prime) {
                System.out.print(i + " ");
                count++;
                if (count % 8 == 0) System.out.println();
            }
        }
    }
}
