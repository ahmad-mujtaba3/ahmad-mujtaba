import java.util.Scanner;

public class Exercise05_41 {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        int max = 0, count = 0, num;
        do {
            num = input.nextInt();
            if (num > max) {
                max = num;
                count = 1;
            } else if (num == max) count++;
        } while (num != 0);
        System.out.println(max + " " + count);
    }
}
