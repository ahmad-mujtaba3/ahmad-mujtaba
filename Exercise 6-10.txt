public class Exercise06_10 {

    public static void main(String[] args) {

        int count = 0;
        int sum = 0;
        for (int i = 1; i <= 100; i++) {
            if (isPrime(i)) {
                sum += i;
                count++;
            }
        }
        System.out.println("Count: " + count);
        System.out.println("Sum: " + sum);
    }
    public static boolean isPrime(int n) {
        if (n < 2) return false;
        for (int i = 2; i <= n / 2; i++)
            if (n % i == 0) return false;
        return true;
    }
}
