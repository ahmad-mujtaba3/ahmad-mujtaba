public class Exercise06_12 {

    public static void main(String[] args) {

        printChars('1', 'Z', 10);
    }
    public static void printChars(char ch1, char ch2, int numberPerLine) {
        int count = 0;
        for (char c = ch1; c <= ch2; c++) {
            System.out.print(c + " ");
            count++;
            if (count % numberPerLine == 0) System.out.println();
        }
    }
}
