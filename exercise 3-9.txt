import java.util.Scanner;
public class Exercise03_09 {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        System.out.print("Enter first 9 digits of ISBN: ");
        int number = input.nextInt();
        int n1 = number / 100000000;
        int n2 = (number / 10000000) % 10;
        int n3 = (number / 1000000) % 10;
        int n4 = (number / 100000) % 10;
        int n5 = (number / 10000) % 10;
        int n6 = (number / 1000) % 10;
        int n7 = (number / 100) % 10;
        int n8 = (number / 10) % 10;
        int n9 = number % 10;
        int checksum = (n1 + n2*2 + n3*3 + n4*4 + n5*5 + n6*6 + n7*7 + n8*8 + n9*9) % 11;
        if (checksum == 10)
            System.out.println(number + "X");
        else
            System.out.println(number + "" + checksum);
    }
}
