import java.util.Scanner;

public class Exercise05_14 {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        int m = input.nextInt();
        int n = input.nextInt();
        int gcd = 1, k = 2;
        while (k <= m && k <= n) {
            if (m % k == 0 && n % k == 0) gcd = k;
            k++;
        }
        System.out.println(gcd);
    }
}
