public class Exercise05_18C {

    public static void main(String[] args) {

        for (int i = 1; i <= 6; i++) {
            for (int s = 6 - i; s > 0; s--) System.out.print("  ");
            for (int j = 1; j <= i; j++)
                System.out.print(j + " ");
            System.out.println();
        }
    }
}
