public class Exercise05_07 {

    public static void main(String[] args) {

        double tuition = 10000;
        for (int i = 0; i < 10; i++)
            tuition *= 1.05;
        double cost = 0;
        for (int i = 0; i < 4; i++) {
            cost += tuition;
            tuition *= 1.05;
        }
        System.out.println(cost);
    }
}
