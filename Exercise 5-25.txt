public class Exercise05_25 {

    public static void main(String[] args) {

        double pi = 0;
        for (long i = 1; i <= 100000; i += 2)
            pi += 4 * ((1.0 / i) - (1.0 / (i + 2)));
        System.out.println(pi);
        pi = 0;
        for (long i = 1; i <= 200000; i += 2)
            pi += 4 * ((1.0 / i) - (1.0 / (i + 2)));
        System.out.println(pi);
        pi = 0;
        for (long i = 1; i <= 300000; i += 2)
            pi += 4 * ((1.0 / i) - (1.0 / (i + 2)));
        System.out.println(pi);
    }
}
