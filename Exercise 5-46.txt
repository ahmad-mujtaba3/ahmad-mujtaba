import java.util.Scanner;

public class Exercise05_46 {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        String s = input.nextLine();
        String r = "";
        for (int i = s.length() - 1; i >= 0; i--)
            r += s.charAt(i);
        System.out.println(r);
    }
}
