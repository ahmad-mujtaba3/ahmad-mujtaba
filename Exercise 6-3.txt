public class Exercise06_03 {

    public static void main(String[] args) {

        for (int n = 1; n <= 200; n++)
            if (isPalindrome(n)) System.out.print(n + " ");
    }
    public static boolean isPalindrome(int n) {
        return n == reverse(n);
    }
    public static int reverse(int n) {
        int rev = 0;
        while (n != 0) {
            rev = rev * 10 + n % 10;
            n /= 10;
        }
        return rev;
    }
}
