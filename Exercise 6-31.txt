import java.util.Scanner;

public class Exercise06_31 {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        long n = input.nextLong();
        System.out.println(isValid(n) ? sumDigits(n) : "Invalid");
    }
    public static boolean isValid(long n) {
        int len = String.valueOf(n).length();
        return len >= 13 && len <= 16 && prefixMatched(n) && (sumOfDoubleEvenPlace(n) + sumOfOddPlace(n)) % 10 == 0;
    }
    public static boolean prefixMatched(long n) {
        return prefix(n, 1) == 4 || prefix(n, 1) == 5 || prefix(n, 1) == 6 || prefix(n, 2) == 37;
    }
    public static int sumOfDoubleEvenPlace(long n) {
        int sum = 0;
        String s = n + "";
        for (int i = s.length() - 2; i >= 0; i -= 2)
            sum += getDigit((s.charAt(i) - '0') * 2);
        return sum;
    }
    public static int sumOfOddPlace(long n) {
        int sum = 0;
        String s = n + "";
        for (int i = s.length() - 1; i >= 0; i -= 2)
            sum += (s.charAt(i) - '0');
        return sum;
    }
    public static int getDigit(int n) {
        return n < 10 ? n : n / 10 + n % 10;
    }
    public static int prefix(long n, int k) {
        String s = n + "";
        return Integer.parseInt(s.substring(0, k));
    }
    public static int sumDigits(long n) {
        int sum = 0;
        while (n != 0) {
            sum += n % 10;
            n /= 10;
        }
        return sum;
    }
}
