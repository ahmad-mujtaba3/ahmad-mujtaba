public class Exercise05_19 {

    public static void main(String[] args) {

        for (int i = 0; i < 8; i++) {
            for (int s = 7 - i; s > 0; s--) System.out.print("   ");
            for (int j = 0; j <= i; j++)
                System.out.printf("%3d", (int)Math.pow(2, j));
            for (int j = i - 1; j >= 0; j--)
                System.out.printf("%3d", (int)Math.pow(2, j));
            System.out.println();
        }
    }
}
