import java.util.Scanner;

public class Exercise05_16 {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int factor = 2;
        while (n > 1) {
            if (n % factor == 0) {
                System.out.print(factor + " ");
                n /= factor;
            } else factor++;
        }
    }
}
