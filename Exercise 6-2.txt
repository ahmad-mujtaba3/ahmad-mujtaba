public class Exercise06_02 {

    public static void main(String[] args) {

        for (int i = 1; i <= 10; i++)
            System.out.println(i + "  " + sumDigits(i));
    }
    public static int sumDigits(long n) {
        int sum = 0;
        while (n != 0) {
            sum += n % 10;
            n /= 10;
        }
        return sum;
    }
}
