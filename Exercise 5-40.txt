public class Exercise05_40 {

    public static void main(String[] args) {

        int heads = 0, tails = 0;
        for (int i = 0; i < 1000000; i++) {
            if ((int)(Math.random() * 2) == 0) heads++;
            else tails++;
        }
        System.out.println("Heads: " + heads);
        System.out.println("Tails: " + tails);
    }
}
