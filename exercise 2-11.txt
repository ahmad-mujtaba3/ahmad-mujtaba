import java.util.Scanner;

public class Exercise02_11 {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        System.out.print("Enter years: ");
        int years = input.nextInt();
        double current = 312032486;
        double seconds = 365 * 24 * 60 * 60;
        double birth = seconds / 7.0;
        double death = seconds / 13.0;
        double immigrant = seconds / 45.0;
        current += years * (birth - death + immigrant);
        System.out.println((int)current);
    }
}
