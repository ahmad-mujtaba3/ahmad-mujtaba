public class Exercise06_40 {

    public static void main(String[] args) {

        for (int i = 1; i <= 10; i++)
            System.out.println(i + "   " + computeFutureTuition(i));
    }
    public static double computeFutureTuition(int year) {
        double tuition = 10000;
        for (int i = 0; i < year; i++)
            tuition *= 1.07;
        return tuition;
    }
}
