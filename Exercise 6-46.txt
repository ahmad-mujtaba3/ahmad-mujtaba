public class Exercise06_46 {

    public static void main(String[] args) {

        System.out.println(count("Mississippi", 's'));
    }
    public static int count(String s, char c) {
        int count = 0;
        for (int i = 0; i < s.length(); i++)
            if (s.charAt(i) == c) count++;
        return count;
    }
}
