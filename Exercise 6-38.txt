public class Exercise06_38 {

    public static void main(String[] args) {

        printRandomLowercase();
        printRandomUppercase();
        printRandomDigit();
    }
    public static void printRandomLowercase() {
        System.out.print((char)('a' + (int)(Math.random() * 26)));
    }
    public static void printRandomUppercase() {
        System.out.print((char)('A' + (int)(Math.random() * 26)));
    }
    public static void printRandomDigit() {
        System.out.print((char)('0' + (int)(Math.random() * 10)));
    }
}
