public class Exercise06_49 {

    public static void main(String[] args) {

        printTriangle(5);
    }
    public static void printTriangle(int n) {
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= i; j++)
                System.out.print(j + " ");
            System.out.println();
        }
    }
}
