public class Exercise04_03 {

    public static void main(String[] args) {

        double x1 = 33.75, y1 = -84.39;
        double x2 = 28.53, y2 = -81.38;
        double x3 = 32.08, y3 = -81.09;
        double x4 = 35.22, y4 = -80.84;
        double x5 = 36.17, y5 = -86.78;
        double d1 = 6371.01 * Math.acos(Math.sin(Math.toRadians(x1)) * Math.sin(Math.toRadians(x2)) +
                Math.cos(Math.toRadians(x1)) * Math.cos(Math.toRadians(x2)) * Math.cos(Math.toRadians(y1 - y2)));
        double d2 = 6371.01 * Math.acos(Math.sin(Math.toRadians(x1)) * Math.sin(Math.toRadians(x3)) +
                Math.cos(Math.toRadians(x1)) * Math.cos(Math.toRadians(x3)) * Math.cos(Math.toRadians(y1 - y3)));
        double d3 = 6371.01 * Math.acos(Math.sin(Math.toRadians(x4)) * Math.sin(Math.toRadians(x5)) +
                Math.cos(Math.toRadians(x4)) * Math.cos(Math.toRadians(x5)) * Math.cos(Math.toRadians(y4 - y5)));
        System.out.println(d1 + d2 + d3);
    }
}
