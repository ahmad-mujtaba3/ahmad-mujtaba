public class Exercise05_23 {

    public static void main(String[] args) {

        double sum = 0;
        for (double i = 1; i <= 50000; i++)
            sum += 1 / i;
        System.out.println(sum);
        sum = 0;
        for (double i = 50000; i >= 1; i--)
            sum += 1 / i;
        System.out.println(sum);
    }
}
