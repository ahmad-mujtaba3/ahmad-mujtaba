import java.util.Scanner;

public class Exercise04_23 {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        System.out.print("Enter employee's name: ");
        String name = input.nextLine();
        System.out.print("Enter number of hours worked: ");
        double hours = input.nextDouble();
        System.out.print("Enter hourly pay rate: ");
        double rate = input.nextDouble();
        System.out.print("Enter federal tax withholding rate: ");
        double fed = input.nextDouble();
        System.out.print("Enter state tax withholding rate: ");
        double state = input.nextDouble();
        double gross = hours * rate;
        double fedTax = gross * fed;
        double stateTax = gross * state;
        double net = gross - fedTax - stateTax;
        System.out.println("Employee Name: " + name);
        System.out.println("Gross Pay: " + gross);
        System.out.println("Deductions:");
        System.out.println("  Federal Withholding (" + (fed * 100) + "%): " + fedTax);
        System.out.println("  State Withholding (" + (state * 100) + "%): " + stateTax);
        System.out.println("  Total Deduction: " + (fedTax + stateTax));
        System.out.println("Net Pay: " + net);
    }
}
